cmake_minimum_required(VERSION 3.15)
project(spike_ci C)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_SHARED_MODULE_PREFIX "")
set(CMAKE_SHARED_LIBRARY_PREFIX "")
add_library(llhttp_lua MODULE c/llhttp_lua.c ${LLHTTP_DIR}/src/api.c ${LLHTTP_DIR}/src/http.c ${LLHTTP_DIR}/src/llhttp.c)
target_include_directories(llhttp_lua PRIVATE ${LUA_INCLUDE_DIR} ${LLHTTP_DIR}/include)
target_link_libraries(llhttp_lua ${LUA_LIBRARIES})
add_library(yy MODULE c/yy.c ${YYJSON_DIR}/src/yyjson.c)
target_include_directories(yy PRIVATE ${LUA_INCLUDE_DIR} ${YYJSON_DIR}/src)
target_link_libraries(yy ${LUA_LIBRARIES})
